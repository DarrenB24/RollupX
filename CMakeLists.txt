cmake_minimum_required(VERSION 3.20)
project(RollupX VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(ROLLUPX_FOUNDATION_ONLY "Build dependency-free foundation checks only (no node)" OFF)
option(ROLLUPX_CORE_ONLY "Build offline crypto/execution and tests without gRPC/RocksDB" OFF)
option(ROLLUPX_LEDGER_ONLY "Build standalone RocksDB ledger without gRPC" OFF)
option(ROLLUPX_WIRE_ONLY "Build protobuf transaction validation without gRPC/RocksDB" OFF)
option(ROLLUPX_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(ROLLUPX_TSAN "Enable thread sanitizer (separate build)" OFF)
if((ROLLUPX_FOUNDATION_ONLY AND ROLLUPX_CORE_ONLY) OR
   (ROLLUPX_LEDGER_ONLY AND (ROLLUPX_FOUNDATION_ONLY OR ROLLUPX_CORE_ONLY)) OR
   (ROLLUPX_WIRE_ONLY AND (ROLLUPX_FOUNDATION_ONLY OR ROLLUPX_CORE_ONLY OR ROLLUPX_LEDGER_ONLY)))
  message(FATAL_ERROR "Choose only one reduced build mode")
endif()
if(ROLLUPX_SANITIZE AND ROLLUPX_TSAN)
  message(FATAL_ERROR "ASan and TSan require separate builds")
endif()
find_package(Threads REQUIRED)
add_library(rollupx_options INTERFACE)
target_compile_options(rollupx_options INTERFACE -Wall -Wextra -Wpedantic -Werror)
if(ROLLUPX_SANITIZE)
  target_compile_options(rollupx_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(rollupx_options INTERFACE -fsanitize=address,undefined)
elseif(ROLLUPX_TSAN)
  target_compile_options(rollupx_options INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(rollupx_options INTERFACE -fsanitize=thread)
endif()

add_library(rollupx_foundation src/node/config.cpp src/metrics/http_server.cpp)
target_include_directories(rollupx_foundation PUBLIC include)
target_link_libraries(rollupx_foundation PUBLIC rollupx_options Threads::Threads)
include(CTest)
if(BUILD_TESTING)
  add_executable(foundation_checks tests/unit/foundation_checks.cpp)
  target_link_libraries(foundation_checks PRIVATE rollupx_foundation)
  add_test(NAME config_checks COMMAND foundation_checks --config-only)
  set_tests_properties(config_checks PROPERTIES TIMEOUT 20 LABELS "unit")
  if(NOT ROLLUPX_CORE_ONLY AND NOT ROLLUPX_LEDGER_ONLY AND NOT ROLLUPX_WIRE_ONLY)
    add_test(NAME http_checks COMMAND foundation_checks)
    set_tests_properties(http_checks PROPERTIES TIMEOUT 20 LABELS "unit;network")
  endif()
endif()
if(ROLLUPX_FOUNDATION_ONLY)
  if(BUILD_TESTING)
    find_package(GTest CONFIG QUIET)
    if(GTest_FOUND)
      add_executable(config_gtests tests/unit/config_test.cpp)
      target_link_libraries(config_gtests PRIVATE rollupx_foundation GTest::gtest_main)
      add_test(NAME config_gtests COMMAND config_gtests)
      set_tests_properties(config_gtests PROPERTIES TIMEOUT 20 LABELS "unit")
    else()
      message(STATUS "GoogleTest not found: only supplementary foundation checks are available")
    endif()
  endif()
  message(STATUS "Foundation only: gRPC, storage, and five-node tests are NOT built")
  return()
endif()

find_package(OpenSSL 3 REQUIRED)
add_library(rollupx_crypto src/crypto/keccak.cpp src/crypto/secp256k1.cpp)
target_include_directories(rollupx_crypto PUBLIC include)
target_link_libraries(rollupx_crypto PUBLIC rollupx_options OpenSSL::Crypto)
add_library(rollupx_execution src/execution/transaction.cpp src/execution/execution.cpp src/storage/ledger_records.cpp)
target_link_libraries(rollupx_execution PUBLIC rollupx_crypto)
add_library(rollupx_raft src/consensus/raft.cpp)
target_include_directories(rollupx_raft PUBLIC include)
target_link_libraries(rollupx_raft PUBLIC rollupx_options)
add_library(rollupx_raft_records src/storage/raft_records.cpp)
target_link_libraries(rollupx_raft_records PUBLIC rollupx_raft rollupx_crypto)
add_library(rollupx_application src/execution/raft_application.cpp)
target_link_libraries(rollupx_application PUBLIC rollupx_execution rollupx_raft_records)
add_executable(rollupx-wallet cli/wallet.cpp)
target_link_libraries(rollupx-wallet PRIVATE rollupx_execution)
install(TARGETS rollupx-wallet RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  find_package(GTest CONFIG REQUIRED)
  add_executable(core_tests tests/unit/crypto_test.cpp tests/unit/execution_test.cpp
    tests/unit/ledger_records_test.cpp tests/unit/recovery_audit_test.cpp)
  target_link_libraries(core_tests PRIVATE rollupx_execution GTest::gtest_main Threads::Threads)
  add_test(NAME core_tests COMMAND core_tests)
  set_tests_properties(core_tests PROPERTIES TIMEOUT 60 LABELS "unit;core")
  add_executable(raft_tests tests/consensus/raft_test.cpp tests/unit/raft_records_test.cpp tests/unit/raft_application_test.cpp)
  target_link_libraries(raft_tests PRIVATE rollupx_application GTest::gtest_main)
  add_test(NAME raft_tests COMMAND raft_tests)
  set_tests_properties(raft_tests PROPERTIES TIMEOUT 30 LABELS "unit;consensus;simulation")
  add_test(NAME wallet_smoke COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/scripts/wallet-smoke.sh $<TARGET_FILE:rollupx-wallet>)
  set_tests_properties(wallet_smoke PROPERTIES TIMEOUT 20 LABELS "integration;core")
endif()
if(ROLLUPX_CORE_ONLY)
  message(STATUS "Core only: no gRPC node, RocksDB, or durable ledger is built")
  return()
endif()

if(NOT ROLLUPX_WIRE_ONLY)
  find_package(RocksDB CONFIG REQUIRED)
  add_library(rollupx_raft_store src/storage/raft_store.cpp)
  target_link_libraries(rollupx_raft_store PUBLIC rollupx_application RocksDB::rocksdb Threads::Threads)
  add_library(rollupx_ledger src/storage/durable_ledger.cpp)
  target_link_libraries(rollupx_ledger PUBLIC rollupx_execution RocksDB::rocksdb Threads::Threads)
  add_executable(rollupx-ledger cli/ledger.cpp)
  target_link_libraries(rollupx-ledger PRIVATE rollupx_ledger)
  install(TARGETS rollupx-ledger RUNTIME DESTINATION bin)
  if(BUILD_TESTING)
    add_executable(raft_store_crash_worker tests/fault_injection/raft_store_crash_worker.cpp)
    target_link_libraries(raft_store_crash_worker PRIVATE rollupx_raft_store)
    add_executable(raft_application_crash_worker tests/fault_injection/raft_application_crash_worker.cpp)
    target_link_libraries(raft_application_crash_worker PRIVATE rollupx_raft_store)
    add_executable(raft_store_tests tests/integration/raft_store_test.cpp tests/integration/replicated_ledger_test.cpp
      tests/fault_injection/raft_store_crash_test.cpp)
    target_link_libraries(raft_store_tests PRIVATE rollupx_raft_store GTest::gtest_main)
    target_compile_definitions(raft_store_tests PRIVATE ROLLUPX_RAFT_CRASH_WORKER="$<TARGET_FILE:raft_store_crash_worker>")
    target_compile_definitions(raft_store_tests PRIVATE ROLLUPX_APPLICATION_CRASH_WORKER="$<TARGET_FILE:raft_application_crash_worker>")
    add_dependencies(raft_store_tests raft_store_crash_worker raft_application_crash_worker)
    add_test(NAME raft_store_tests COMMAND raft_store_tests)
    set_tests_properties(raft_store_tests PROPERTIES TIMEOUT 300 LABELS "integration;storage;fault;consensus")
    add_executable(ledger_crash_worker tests/fault_injection/ledger_crash_worker.cpp)
    target_link_libraries(ledger_crash_worker PRIVATE rollupx_ledger)
    add_executable(ledger_tests tests/integration/durable_ledger_test.cpp tests/fault_injection/ledger_crash_test.cpp)
    target_link_libraries(ledger_tests PRIVATE rollupx_ledger GTest::gtest_main)
    target_compile_definitions(ledger_tests PRIVATE ROLLUPX_CRASH_WORKER="$<TARGET_FILE:ledger_crash_worker>")
    add_dependencies(ledger_tests ledger_crash_worker)
    add_test(NAME ledger_tests COMMAND ledger_tests)
    set_tests_properties(ledger_tests PROPERTIES TIMEOUT 150 LABELS "integration;storage;fault")
    add_test(NAME ledger_smoke COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/scripts/ledger-smoke.sh
      $<TARGET_FILE:rollupx-ledger> $<TARGET_FILE:rollupx-wallet>)
    set_tests_properties(ledger_smoke PROPERTIES TIMEOUT 30 LABELS "integration;storage")
  endif()
  if(ROLLUPX_LEDGER_ONLY)
    message(STATUS "Standalone DEVELOPMENT ledger: no gRPC or Raft consensus")
    return()
  endif()
endif()

find_package(Protobuf CONFIG QUIET)
if(NOT Protobuf_FOUND)
  find_package(Protobuf REQUIRED)
endif()
set(GENERATED_DIR ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${GENERATED_DIR})
set(PROTO_NAMES transaction consensus ledger)
foreach(NAME IN LISTS PROTO_NAMES)
  set(PROTO ${CMAKE_CURRENT_SOURCE_DIR}/proto/${NAME}.proto)
  list(APPEND MESSAGE_SOURCES ${GENERATED_DIR}/${NAME}.pb.cc)
  add_custom_command(
    OUTPUT ${GENERATED_DIR}/${NAME}.pb.cc ${GENERATED_DIR}/${NAME}.pb.h
    COMMAND protobuf::protoc
    ARGS --proto_path=${CMAKE_CURRENT_SOURCE_DIR}/proto --cpp_out=${GENERATED_DIR}
         ${PROTO}
    DEPENDS ${PROTO} ${CMAKE_CURRENT_SOURCE_DIR}/proto/transaction.proto
    VERBATIM)
endforeach()
add_library(rollupx_messages ${MESSAGE_SOURCES})
target_include_directories(rollupx_messages SYSTEM PUBLIC ${GENERATED_DIR})
target_link_libraries(rollupx_messages PUBLIC protobuf::libprotobuf)
add_library(rollupx_wire src/networking/transaction_wire.cpp)
target_link_libraries(rollupx_wire PUBLIC rollupx_execution rollupx_messages)
add_executable(rollupx-wire cli/wire.cpp)
target_link_libraries(rollupx-wire PRIVATE rollupx_wire)
install(TARGETS rollupx-wire RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  add_executable(wire_tests tests/unit/transaction_wire_test.cpp)
  target_link_libraries(wire_tests PRIVATE rollupx_wire GTest::gtest_main)
  add_test(NAME wire_tests COMMAND wire_tests)
  set_tests_properties(wire_tests PROPERTIES TIMEOUT 30 LABELS "unit;wire")
  add_test(NAME wire_smoke COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/scripts/wire-smoke.sh
    $<TARGET_FILE:rollupx-wallet> $<TARGET_FILE:rollupx-wire>)
  set_tests_properties(wire_smoke PROPERTIES TIMEOUT 20 LABELS "integration;wire")
endif()
if(ROLLUPX_WIRE_ONLY)
  message(STATUS "Wire validation only: no gRPC node or durable database")
  return()
endif()

find_package(gRPC CONFIG REQUIRED)
foreach(NAME IN LISTS PROTO_NAMES)
  set(PROTO ${CMAKE_CURRENT_SOURCE_DIR}/proto/${NAME}.proto)
  list(APPEND GRPC_SOURCES ${GENERATED_DIR}/${NAME}.grpc.pb.cc)
  add_custom_command(
    OUTPUT ${GENERATED_DIR}/${NAME}.grpc.pb.cc ${GENERATED_DIR}/${NAME}.grpc.pb.h
    COMMAND protobuf::protoc
    ARGS --proto_path=${CMAKE_CURRENT_SOURCE_DIR}/proto --grpc_out=${GENERATED_DIR}
         --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin> ${PROTO}
    DEPENDS ${PROTO} ${CMAKE_CURRENT_SOURCE_DIR}/proto/transaction.proto
    VERBATIM)
endforeach()
add_library(rollupx_proto ${GRPC_SOURCES})
target_link_libraries(rollupx_proto PUBLIC rollupx_messages gRPC::grpc++)
add_library(rollupx_storage src/storage/identity_store.cpp)
target_include_directories(rollupx_storage PUBLIC include)
target_link_libraries(rollupx_storage PUBLIC rollupx_foundation RocksDB::rocksdb)
add_library(rollupx_network src/networking/node.cpp)
target_include_directories(rollupx_network PUBLIC include)
target_link_libraries(rollupx_network PUBLIC rollupx_foundation rollupx_storage rollupx_proto)
add_executable(rollupx-node src/node/main.cpp)
target_link_libraries(rollupx-node PRIVATE rollupx_network)
add_executable(rollupx-cli cli/main.cpp)
target_link_libraries(rollupx-cli PRIVATE rollupx_proto rollupx_options)
install(TARGETS rollupx-node rollupx-cli RUNTIME DESTINATION bin)

if(BUILD_TESTING)
  find_package(GTest CONFIG REQUIRED)
  add_executable(rollupx_tests tests/unit/config_test.cpp tests/unit/storage_test.cpp tests/integration/network_test.cpp)
  target_link_libraries(rollupx_tests PRIVATE rollupx_network GTest::gtest_main)
  add_test(NAME rollupx_tests COMMAND rollupx_tests)
  set_tests_properties(rollupx_tests PROPERTIES TIMEOUT 90 LABELS "unit;integration")
  add_test(NAME five_process_mesh COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/scripts/local-smoke.sh
           $<TARGET_FILE:rollupx-node> $<TARGET_FILE:rollupx-cli>)
  set_tests_properties(five_process_mesh PROPERTIES TIMEOUT 90 LABELS "integration")
endif()
